#!/usr/bin/env bash
# Single-run pull-request contract: title, do-not-merge label, issue linkage,
# and, when `enforce-hold-label` is on, the hold label.
#
# One composite step, so a consumer needs exactly one required check
# (`ci-status`). Every check reads the LIVE pull request from the API rather than the event
# payload, so `edited`, `labeled` and `unlabeled` runs see current state.
set -euo pipefail

: "${REPOSITORY:?REPOSITORY is required}"
: "${TYPES:?TYPES is required}"

PR_NUMBER="${PR_NUMBER:-}"
REQUIRE_SCOPE="${REQUIRE_SCOPE:-false}"
DO_NOT_MERGE_LABEL="${DO_NOT_MERGE_LABEL:-do-not-merge}"
EXEMPT_AUTHORS="${EXEMPT_AUTHORS:-}"
LINKAGE_LABEL="${LINKAGE_LABEL:-needs-issue-linkage}"
LINKAGE_MODE="${LINKAGE_MODE:-advisory}"
ENFORCE_HOLD_LABEL="${ENFORCE_HOLD_LABEL:-false}"
HOLD_LABEL="${HOLD_LABEL:-needs-human}"
# GitHub's own grammar: `#N` with an optional `OWNER/REPO` prefix.
DEFAULT_ISSUE_REF_PATTERN='([a-z0-9_.-]+/[a-z0-9_.-]+)?#[0-9]+'
ISSUE_REF_PATTERN="${ISSUE_REF_PATTERN:-$DEFAULT_ISSUE_REF_PATTERN}"
export ISSUE_REF_PATTERN DEFAULT_ISSUE_REF_PATTERN

# Stable HTML marker on the advisory comment: the upsert finds its own previous
# comment by this string, so a re-run edits rather than posting a second one.
LINKAGE_MARKER='<!-- check-contract:linkage -->'

scratch="$(mktemp -d)"
trap 'declare -F gh_api_calls_report >/dev/null && gh_api_calls_report; rm -rf -- "$scratch"' EXIT
gh_stdout="$scratch/gh-stdout"
gh_stderr="$scratch/gh-stderr"
gh_headers="$scratch/gh-headers"
# shellcheck source=../gh-api.sh
source "${BASH_SOURCE[0]%/*}/../gh-api.sh"

emit_output() {
  local name="$1" value="$2"
  if [[ -n "${GITHUB_OUTPUT:-}" ]]; then
    echo "${name}=${value}" >>"$GITHUB_OUTPUT"
  fi
}

# emit_results <title> <do-not-merge> <linkage> <hold>
emit_results() {
  emit_output title "$1"
  emit_output do-not-merge "$2"
  emit_output linkage "$3"
  emit_output hold "$4"
}

# Every annotation below quotes attacker-controlled text: the PR title, the PR
# body, the author login. GitHub's documented escaping for workflow-command data
# is `%` -> `%25`, CR -> `%0D`, LF -> `%0A`; without it a title carrying a
# newline can close the annotation and inject a second workflow command.
# `%` first, or the escapes introduced by the others get double-escaped.
escape_annotation() {
  local text="$1"
  text="${text//'%'/%25}"
  text="${text//$'\r'/%0D}"
  text="${text//$'\n'/%0A}"
  printf '%s' "$text"
}

# Every value below is interpolated into a `gh api` path. Validate before the
# first call rather than trusting the caller's expression.
require_pattern() {
  local name="$1" value="$2" pattern="$3" shape="$4"
  if [[ ! "$value" =~ $pattern ]]; then
    echo "::error::check-contract: ${name} must be ${shape}, got: $(escape_annotation "$value")"
    exit 1
  fi
}

# Percent-encode a label so it reaches the DELETE path as one segment. `@uri`
# encodes UTF-8 bytes, so an emoji or non-ASCII label survives.
# shellcheck disable=SC2329 # invoked from remove_linkage_label, itself reached through best_effort.
url_encode() {
  jq -rn --arg text "$1" '$text|@uri'
}

require_pattern linkage-mode "$LINKAGE_MODE" '^(advisory|enforce)$' "'advisory' or 'enforce'"
require_pattern require-scope "$REQUIRE_SCOPE" '^(true|false)$' "'true' or 'false'"
require_pattern enforce-hold-label "$ENFORCE_HOLD_LABEL" '^(true|false)$' "'true' or 'false'"

hold_result=skipped

# The pattern reaches awk only through ENVIRON, never the program text, so it
# cannot inject awk code. It must compile both bare and as the parenthesized
# group the analyzer splices it into (an unbalanced `)` would otherwise escape
# the group), and must not match the empty string, which would make a bare
# "Closes" count as a closing reference.
if [[ "$ISSUE_REF_PATTERN" == *$'\n'* || "$ISSUE_REF_PATTERN" == *$'\r'* ]]; then
  echo '::error::check-contract: issue-ref-pattern must be a single line'
  exit 1
fi
# Spliced after a keyword, an anchor could never match mid-line.
if [[ "$ISSUE_REF_PATTERN" == '^'* || ("$ISSUE_REF_PATTERN" == *'$' && "$ISSUE_REF_PATTERN" != *'\$') ]]; then
  echo "::error::check-contract: issue-ref-pattern must not be anchored with ^ or \$: $(escape_annotation "$ISSUE_REF_PATTERN")"
  exit 1
fi
# Case-insensitive matching lowercases the body and the pattern with the same
# awk `tolower`. A case class name is already lowercase, so `[:upper:]` would
# then match nothing; it becomes `[:alpha:]`, its case-folded equivalent. This
# fold is repeated verbatim in analyze_body.
pattern_status=0
awk 'BEGIN {
  p = tolower(ENVIRON["ISSUE_REF_PATTERN"])
  gsub(/\[:upper:\]/, "[:alpha:]", p)
  if ("" ~ p || "" ~ ("(" p ")")) exit 3
}' </dev/null 2>/dev/null || pattern_status=$?
if [[ "$pattern_status" -ne 0 ]]; then
  if [[ "$pattern_status" -eq 3 ]]; then
    echo "::error::check-contract: issue-ref-pattern matches the empty string: $(escape_annotation "$ISSUE_REF_PATTERN")"
  else
    echo "::error::check-contract: issue-ref-pattern is not a valid extended regex: $(escape_annotation "$ISSUE_REF_PATTERN")"
  fi
  exit 1
fi

# Step 0 — a merge group carries no pull request in its payload; the queued
# one is named in the group's head ref, `gh-readonly-queue/<base>/pr-<N>-<sha>`.
# An unparsable ref fails closed: a group whose pull request cannot be named
# cannot be checked for the do-not-merge label.
if [[ "${EVENT_NAME:-}" == merge_group ]]; then
  merge_group_ref_pattern='^(refs/heads/)?gh-readonly-queue/(.+)/pr-([0-9]+)-[0-9a-f]{40}$'
  if [[ ! "${MERGE_GROUP_HEAD_REF:-}" =~ $merge_group_ref_pattern ]]; then
    echo "::error::check-contract: cannot read a pull request number from the merge-group head ref '$(escape_annotation "${MERGE_GROUP_HEAD_REF:-}")'"
    if [[ "$ENFORCE_HOLD_LABEL" == true ]]; then
      hold_result=fail
    fi
    emit_results skipped fail skipped "$hold_result"
    exit 1
  fi
  # The queue's branch reaches only a GraphQL variable, so it is not validated.
  merge_group_branch="${BASH_REMATCH[2]}"
  PR_NUMBER="${BASH_REMATCH[3]}"
# A push, schedule or workflow_dispatch run has no pull request at all. The
# composite is a pull-request gate; such a run must report skipped, not fail.
elif [[ -z "${PR_NUMBER//[[:space:]]/}" ]]; then
  echo '::notice::check-contract: no pull request in this event; nothing to check'
  emit_results skipped skipped skipped skipped
  exit 0
fi

# Only the two values that reach an API path unencoded. The labels are
# deliberately NOT validated: GitHub labels may legally contain spaces and
# colons (`do not merge`, `status: blocked`), and neither one is ever
# interpolated raw — the blocking label is compared with `grep -qxF`, the
# linkage label travels in a JSON body built by `jq --arg`, and the only path
# it reaches is percent-encoded below.
require_pattern repository "$REPOSITORY" '^[A-Za-z0-9_.-]+/[A-Za-z0-9_.-]+$' 'OWNER/REPO'
require_pattern pr-number "$PR_NUMBER" '^[0-9]+$' 'a positive integer'

# At most an hour, the life of a GitHub App installation token.
require_pattern rate-limit-wait-seconds "${RATE_LIMIT_WAIT_SECONDS:-60}" '^(0|[1-9][0-9]{0,2}|[1-2][0-9]{3}|3[0-5][0-9]{2}|3600)$' 'an integer number of seconds from 0 to 3600'

# Best-effort write: a refused or missing write surfaces as a ::notice:: and
# never changes the exit code. The advisory comment and label are convenience,
# not the gate.
best_effort() {
  local description="$1"
  shift
  if ! "$@"; then
    echo "::notice::check-contract: ${description} was refused (HTTP ${GH_HTTP_STATUS:-unknown}); continuing without it"
  fi
}

# ---------------------------------------------------------------------------
# Step 1 — fetch the pull request, and its comments for the marker upsert,
# in one GraphQL query. GraphQL, not `pulls/N`: the REST bucket is the one
# that runs out during a burst of pull request CI.
# ---------------------------------------------------------------------------
# shellcheck disable=SC2016 # `$owner`, `$name` and `$number` are GraphQL variables.
pr_query='query($owner: String!, $name: String!, $number: Int!) { repository(owner: $owner, name: $name) { pullRequest(number: $number) { title body author { __typename login } labels(first: 100) { totalCount nodes { name } } comments(last: 100) { totalCount nodes { databaseId body author { __typename } } } } } }'
pr_read="pull request ${REPOSITORY}#${PR_NUMBER}"
pr_query_variables=(-f owner="${REPOSITORY%%/*}" -f name="${REPOSITORY#*/}" -F number="$PR_NUMBER")
# A merge group lands every queued entry up to its own in one merge, so the
# hold is read for each of them, in the same request: the queue's entries and
# their labels. 100 entries of 100 labels stays far inside GraphQL's node limit.
if [[ "$ENFORCE_HOLD_LABEL" == true && "${EVENT_NAME:-}" == merge_group ]]; then
  # shellcheck disable=SC2016 # `$owner`, `$name`, `$number` and `$branch` are GraphQL variables.
  pr_query='query($owner: String!, $name: String!, $number: Int!, $branch: String!) { repository(owner: $owner, name: $name) { pullRequest(number: $number) { title body author { __typename login } labels(first: 100) { totalCount nodes { name } } comments(last: 100) { totalCount nodes { databaseId body author { __typename } } } } mergeQueue(branch: $branch) { entries(first: 100) { totalCount nodes { position pullRequest { number labels(first: 100) { totalCount nodes { name } } } } } } } }'
  pr_query_variables+=(-f branch="$merge_group_branch")
fi
# Named on every failure below, so a red that blocks a held pull request says
# why it is red and that a re-run is what clears it.
hold_unreadable() {
  if [[ "$ENFORCE_HOLD_LABEL" == true ]]; then
    hold_result=fail
    echo "::error::check-contract: hold-label-unreadable: cannot tell whether $1 carries the '$(escape_annotation "$HOLD_LABEL")' hold label, so this run fails; re-run this job once GitHub answers."
  fi
}
fail_pr_unreadable() {
  hold_unreadable "$pr_read"
  emit_output hold "$hold_result"
  exit 1
}
# shellcheck disable=SC2310 # gh_api handles its own errexit; the caller classifies the status.
if ! gh_api POST graphql -f query="$pr_query" "${pr_query_variables[@]}"; then
  cat "$gh_stderr" >&2
  if [[ "$GH_FAILURE" == rate-limit ]]; then
    echo "::error::check-contract: could not read ${pr_read} (HTTP ${GH_HTTP_STATUS}): rate limited past rate-limit-wait-seconds; re-run this job after the limit resets"
  else
    echo "::error::check-contract: could not read ${pr_read} (HTTP ${GH_HTTP_STATUS:-unknown}): $(escape_annotation "$GH_MESSAGE")"
  fi
  fail_pr_unreadable
fi
pr_response="$scratch/pr-response.json"
cp -- "$gh_stdout" "$pr_response"
pr_json="$scratch/pr.json"
# Every field a verdict reads must be present with its type, or the read fails
# closed: GraphQL answers a pull request it cannot resolve with `null` and no
# error, and a `labels` page that does not hold every label could hide
# `do-not-merge`. The pull request is rewritten into the REST shape the checks
# below read. A Bot's GraphQL login lacks the `[bot]` suffix its REST login
# carries (`dependabot` against `dependabot[bot]`), so it is added back for
# `exempt-authors`. A deleted author reads `null`, which exempts nobody.
if ! jq -e '.data.repository.pullRequest
  | select(type == "object" and (.title | type) == "string" and (.body | type) == "string"
      and (.labels.totalCount | type) == "number" and (.labels.nodes | type) == "array"
      and (.labels.totalCount == (.labels.nodes | length))
      and all(.labels.nodes[]; (.name | type) == "string"))
  | {title, body,
     user: {login: (if .author == null then "" elif .author.__typename == "Bot" then "\(.author.login)[bot]" else (.author.login // "") end)},
     labels: [.labels.nodes[] | {name}]}' <"$gh_stdout" >"$pr_json"; then
  echo "::error::check-contract: ${pr_read} read over GraphQL lacks its title, body or complete label list; failing closed"
  fail_pr_unreadable
fi
# The comments, in the REST listing's shape, when the one page holds them all.
# Otherwise the marker lookup lists them over REST.
comments_json="$scratch/comments.json"
jq -e '.data.repository.pullRequest.comments
  | select((.totalCount | type) == "number" and (.nodes | type) == "array" and .totalCount == (.nodes | length)
      and all(.nodes[]; (.databaseId | type) == "number" and (.body | type) == "string"))
  | [[.nodes[] | {id: .databaseId, user: {type: (.author.__typename // "")}, body}]]' \
  <"$gh_stdout" >"$comments_json" 2>/dev/null || rm -f -- "$comments_json"

pr_title="$(jq -r .title <"$pr_json")"
pr_author="$(jq -r .user.login <"$pr_json")"
jq -r .body <"$pr_json" >"$scratch/body.txt"
jq -r '.labels[].name' <"$pr_json" >"$scratch/labels.txt"

# ---------------------------------------------------------------------------
# Step 2 — title against Conventional Commits with the caller's type list.
# ---------------------------------------------------------------------------
types_list="$(printf '%s' "$TYPES" | tr ',' '\n' | sed 's/^[[:space:]]*//; s/[[:space:]]*$//' | grep -v '^$' || true)"
if [[ -z "$types_list" ]]; then
  echo "::error::check-contract: types is empty; list at least one allowed Conventional Commits type"
  exit 1
fi
types_alternation="$(printf '%s' "$types_list" | paste -sd '|' -)"
types_human="${types_alternation//|/, }"

# `: +`, not `: `: the conventional-commits parser tolerates more than one space
# after the colon.
# shellcheck disable=SC2016 # the backticks are Markdown code spans in a message, not command substitution.
if [[ "$REQUIRE_SCOPE" == true ]]; then
  title_regex="^(${types_alternation})\([^)]+\)!?: +[^[:space:]]"
  scope_hint='a scope is REQUIRED, e.g. `feat(api): add the widget endpoint`'
else
  title_regex="^(${types_alternation})(\([^)]+\))?!?: +[^[:space:]]"
  scope_hint='an optional scope may follow the type, e.g. `feat(api): add the widget endpoint`'
fi

title_result=pass
if [[ ! "$pr_title" =~ $title_regex ]]; then
  title_result=fail
  echo "::error::check-contract: the PR title does not follow Conventional Commits: \"$(escape_annotation "$pr_title")\". Expected \"<type>: <subject>\" with a non-empty subject; allowed types: ${types_human}. ${scope_hint} A breaking change marks the type with \"!\"."
fi

# ---------------------------------------------------------------------------
# Step 3 — do-not-merge label.
# ---------------------------------------------------------------------------
do_not_merge_result=pass
if grep -qxF -- "$DO_NOT_MERGE_LABEL" "$scratch/labels.txt"; then
  do_not_merge_result=fail
  echo "::error::check-contract: this PR carries the '${DO_NOT_MERGE_LABEL}' label; remove it to merge."
fi

# ---------------------------------------------------------------------------
# Step 3b — the hold label, when `enforce-hold-label` is on. Only a person
# removing the label releases the hold. On a merge group it is read for every
# queue entry at or ahead of the queued pull request: the group holds their
# changes, and a passing group merges them all, whatever their own groups
# read earlier. Lower `position` is nearer the front.
# ---------------------------------------------------------------------------
if [[ "$ENFORCE_HOLD_LABEL" == true ]]; then
  if [[ "${EVENT_NAME:-}" != merge_group ]]; then
    hold_result=pass
    if grep -qxF -- "$HOLD_LABEL" "$scratch/labels.txt"; then
      hold_result=fail
      echo "::error::check-contract: this PR carries the '$(escape_annotation "$HOLD_LABEL")' hold label; a person removes it to release the hold."
    fi
  else
    # `held <numbers>` or `clear` when the queue was read whole; empty when it
    # was not, which fails closed.
    # shellcheck disable=SC2016 # `$number` and `$label` are jq variables.
    hold_verdict="$(jq -r --argjson number "$PR_NUMBER" --arg label "$HOLD_LABEL" '
      .data.repository.mergeQueue.entries
      | select(type == "object" and (.totalCount | type) == "number" and (.nodes | type) == "array"
          and .totalCount == (.nodes | length)
          and all(.nodes[]; (.position | type) == "number" and (.pullRequest | type) == "object"
            and (.pullRequest.number | type) == "number"
            and (.pullRequest.labels.totalCount | type) == "number" and (.pullRequest.labels.nodes | type) == "array"
            and .pullRequest.labels.totalCount == (.pullRequest.labels.nodes | length)
            and all(.pullRequest.labels.nodes[]; (.name | type) == "string")))
      | .nodes as $entries
      | [$entries[] | select(.pullRequest.number == $number)]
      | select(length == 1)
      | .[0].position as $own
      | [$entries[] | select(.position <= $own and any(.pullRequest.labels.nodes[]; .name == $label)) | .pullRequest.number] | sort
      | if length == 0 then "clear" else "held \(map(tostring) | join(" "))" end
    ' <"$pr_response" 2>/dev/null || true)"
    case "$hold_verdict" in
      clear) hold_result=pass ;;
      held\ *)
        hold_result=fail
        read -ra held_numbers <<<"${hold_verdict#held }"
        for held_number in "${held_numbers[@]}"; do
          echo "::error::check-contract: queued pull request #${held_number} carries the '$(escape_annotation "$HOLD_LABEL")' hold label and is in this merge group; a person removes it to release the hold."
        done
        ;;
      *) hold_unreadable "the merge queue of '$(escape_annotation "$merge_group_branch")' (pull request #${PR_NUMBER} absent from it, or its entries or their labels incomplete)" ;;
    esac
  fi
fi

# ---------------------------------------------------------------------------
# Merge group — the title and the live do-not-merge label, both re-read because
# either can change after enqueue; a failure makes the queue drop the pull
# request. Linkage is advisory and a group run writes no comment or label.
# ---------------------------------------------------------------------------
if [[ "${EVENT_NAME:-}" == merge_group ]]; then
  emit_results "$title_result" "$do_not_merge_result" skipped "$hold_result"
  if [[ "$title_result" == fail || "$do_not_merge_result" == fail || "$hold_result" == fail ]]; then
    echo "::error::check-contract: queued pull request #${PR_NUMBER} fails the contract; the merge queue removes it. Fix it and re-queue to merge."
    exit 1
  fi
  echo "check-contract: queued pull request #${PR_NUMBER} passes the title and '${DO_NOT_MERGE_LABEL}' checks."
  if [[ "$hold_result" == pass ]]; then
    echo "check-contract: no pull request in this merge group carries the '${HOLD_LABEL}' hold label."
  fi
  exit 0
fi

# ---------------------------------------------------------------------------
# Step 4 — issue linkage.
#
# The body must
# carry a native closing keyword, an explicit non-closing `Refs:`/`Relates to:`
# marker, or a no-issue opt-out, AND four non-empty contract sections. A negated
# closing reference fails outright and is never excused by a valid marker
# elsewhere, because GitHub's own parser is negation-blind and closes the issue
# on merge regardless of the disclaimer.
#
# The analyzer masks rendered HTML comments, fenced and indented code blocks,
# and inline code spans before matching, so a PR template's commented-out
# example cannot satisfy the gate.
# ---------------------------------------------------------------------------
analyze_body() {
  awk '
# Whitespace includes newlines: section content is joined before trimming, so a
# section holding only blank lines must trim to the empty string.
function trim(s) {
  sub(/^[ \t\r\n]+/, "", s)
  sub(/[ \t\r\n]+$/, "", s)
  return s
}

function run_length(s, ch,   n) {
  n = 0
  while (substr(s, n + 1, 1) == ch) n++
  return n
}

# A backtick run of the same length later on this line closes an inline span.
# The upstream implementation also scans following lines; a multi-line inline
# span in a PR body is not a shape this gate needs to model.
function has_closing_run(s, start, ticks,   col, stop) {
  col = start
  while (col <= length(s)) {
    if (substr(s, col, 1) != "`") {
      col++
      continue
    }
    stop = col + 1
    while (substr(s, stop, 1) == "`") stop++
    if (stop - col == ticks) return 1
    col = stop
  }
  return 0
}

# The negation window is the last five words before the keyword, cut at the
# nearest sentence punctuation. The backward scan stops at whichever comes
# first, so each call costs the window, not the whole preceding line; copying
# the preceding text per keyword made a long run of keywords super-quadratic.
function negation_trigger(line, keyword_index,   start, i, ch, width, in_word, has_letter, seen, tail, count, words, word, lower) {
  in_word = 0
  has_letter = 0
  seen = 0
  i = keyword_index - 1
  while (i >= 1) {
    ch = substr(line, i, 1)
    width = 0
    if (ch ~ /[A-Za-z]/) {
      width = 1
      has_letter = 1
    } else if (ch == "'"'"'" || ch == apostrophe) {
      width = 1
    } else if (i >= 3 && substr(line, i - 2, 3) == apostrophe) {
      # A byte-oriented awk sees the typographic apostrophe as three characters.
      width = 3
    }
    if (width) {
      in_word = 1
      i -= width
      continue
    }
    # A word is a run of letters and apostrophes holding at least one letter.
    if (in_word && has_letter && ++seen == 5) break
    in_word = 0
    has_letter = 0
    if (ch == "." || ch == "!" || ch == "?" || ch == ";" || ch == ",") break
    i--
  }
  start = i + 1
  tail = substr(line, start, keyword_index - start)
  # Normalize a typographic apostrophe to the straight one so
  # "doesn<U+2019>t close #N" is detected like "doesn'"'"'t".
  gsub(apostrophe, "'"'"'", tail)
  count = 0
  while (match(tail, /[A-Za-z][A-Za-z'"'"']*/)) {
    count++
    words[count] = substr(tail, RSTART, RLENGTH)
    tail = substr(tail, RSTART + RLENGTH)
  }
  for (i = 1; i <= count; i++) {
    word = words[i]
    lower = tolower(word)
    # Correlative "not only ... but" is affirmative, not a disclaimer.
    if (lower == "not" && i < count && tolower(words[i + 1]) == "only") continue
    if (lower == "not" || lower == "never" || lower == "no" || lower == "without" ||
        lower == "deliberately" || lower == "intentionally") return word
    if (tolower(substr(word, length(word) - 2)) == "n'"'"'t") return word
  }
  return ""
}

function scan_line(line,   indent, rest, lower, marked, pieces, count, i, position, start, len, before, after, text, trigger) {
  indent = 0
  while (substr(line, indent + 1, 1) == " ") indent++
  if (indent <= 3) {
    rest = tolower(substr(line, indent + 1))
    if (rest ~ non_closing_regex) {
      has_non_closing = 1
    }
  }
  lower = tolower(line)
  # One gsub pass finds the same successive leftmost-longest matches as calling
  # match() on each remainder, without rescanning the rest of the line per
  # match. A record never holds a newline, so it can fence each match; the
  # pieces then alternate text, match, text.
  marked = lower
  gsub(closing_regex, "\n&\n", marked)
  count = split(marked, pieces, "\n")
  position = 1
  for (i = 1; i < count; i += 2) {
    start = position + length(pieces[i])
    len = length(pieces[i + 1])
    position = start + len
    before = (start > 1) ? substr(lower, start - 1, 1) : ""
    after = substr(lower, start + len, 1)
    # Word-boundary equivalent of the upstream /\b...\b/ anchors.
    if (before ~ /[a-z0-9_]/) continue
    if (after ~ /[a-z0-9_]/) continue
    # A custom id can contain hyphens, so "ENG-123-extra" is not a complete
    # ENG-123; `#N` keeps the original boundary.
    if (custom && after == "-") continue
    text = substr(line, start, len)
    trigger = negation_trigger(line, start)
    if (trigger != "") {
      if (!(text in negated_trigger)) {
        negated_count++
        negated_order[negated_count] = text
        negated_trigger[text] = trigger
      }
    } else {
      has_closing = 1
    }
  }
}

function section_report(name,   i, trimmed, start, hashes, next_char, tail, content) {
  start = 0
  for (i = 1; i <= line_count; i++) {
    trimmed = tolower(trim(masked[i]))
    if (trimmed ~ ("^##[ \t]+" tolower(name) "$")) {
      start = i
      break
    }
  }
  if (start == 0) {
    print "section-missing\t" name
    return
  }
  content = ""
  for (i = start + 1; i <= line_count; i++) {
    trimmed = trim(masked[i])
    hashes = run_length(trimmed, "#")
    if (hashes >= 1 && hashes <= 6) {
      next_char = substr(trimmed, hashes + 1, 1)
      tail = trim(substr(trimmed, hashes + 1))
      # Only a heading at the same or higher level ends the section, so a
      # nested subsection counts as this section content.
      if ((next_char == " " || next_char == "\t") && tail != "" && hashes <= 2) break
    }
    content = content masked[i] "\n"
  }
  if (trim(content) == "") print "section-empty\t" name
}

BEGIN {
  line_count = 0
  comment_open = 0
  inline_ticks = 0
  fence_char = ""
  fence_len = 0
  has_closing = 0
  has_non_closing = 0
  negated_count = 0
  # U+2019 RIGHT SINGLE QUOTATION MARK, written as its UTF-8 bytes so the source
  # of this action stays plain ASCII.
  apostrophe = "\342\200\231"
  # The caller pattern arrives through ENVIRON, so awk applies no escape
  # processing to it and it never becomes program text.
  ref = tolower(ENVIRON["ISSUE_REF_PATTERN"])
  gsub(/\[:upper:\]/, "[:alpha:]", ref)
  ref = "(" ref ")"
  custom = (ENVIRON["ISSUE_REF_PATTERN"] != ENVIRON["DEFAULT_ISSUE_REF_PATTERN"])
  non_closing_regex = "^(refs|relates[ \t]+to):[ \t]*" ref "[ \t]*$"
  # A custom id can start with letters, so the keyword needs a real separator,
  # spellchecker:ignore-next-line
  # or "fixture-1" would read as "fix" + "ture-1". `#N` cannot continue a word,
  # so the default keeps the original optional separator exactly.
  separator = custom ? "([ \t]+|[ \t]*:[ \t]*)" : "[ \t]*:?[ \t]*"
  closing_regex = "(close|closes|closed|fix|fixes|fixed|resolve|resolves|resolved)" separator ref
}

{
  line = $0
  sub(/\r$/, "", line)

  if (!comment_open && inline_ticks == 0) {
    indent = 0
    while (substr(line, indent + 1, 1) == " ") indent++
    rest = substr(line, indent + 1)
    marker_char = substr(rest, 1, 1)
    marker_run = 0
    if (marker_char == "`" || marker_char == "~") marker_run = run_length(rest, marker_char)
    is_marker = (indent <= 3 && marker_run >= 3)
    info = is_marker ? substr(rest, marker_run + 1) : ""

    if (fence_char != "") {
      masked[++line_count] = ""
      if (is_marker && marker_char == fence_char && marker_run >= fence_len && info ~ /^[ \t]*$/) {
        fence_char = ""
        fence_len = 0
      }
      next
    }
    if (is_marker && !(marker_char == "`" && index(info, "`") > 0)) {
      fence_char = marker_char
      fence_len = marker_run
      masked[++line_count] = ""
      next
    }
    if (substr(line, 1, 4) == "    " || substr(line, 1, 1) == "\t") {
      masked[++line_count] = ""
      next
    }
  }

  rendered = ""
  index_position = 1
  line_length = length(line)
  while (index_position <= line_length) {
    if (comment_open) {
      close_at = index(substr(line, index_position), "-->")
      if (close_at == 0) {
        index_position = line_length + 1
        continue
      }
      comment_open = 0
      index_position = index_position + close_at - 1 + 3
      continue
    }
    character = substr(line, index_position, 1)
    if (character == "`") {
      stop = index_position + 1
      while (substr(line, stop, 1) == "`") stop++
      ticks = stop - index_position
      was_inline = (inline_ticks != 0)
      if (!was_inline && has_closing_run(line, stop, ticks)) {
        inline_ticks = ticks
      } else if (inline_ticks == ticks) {
        inline_ticks = 0
      }
      if (!was_inline && inline_ticks == 0) rendered = rendered substr(line, index_position, stop - index_position)
      index_position = stop
      continue
    }
    if (inline_ticks == 0 && substr(line, index_position, 4) == "<!--") {
      comment_open = 1
      index_position += 4
      continue
    }
    if (inline_ticks == 0) rendered = rendered character
    index_position++
  }
  masked[++line_count] = rendered
}

END {
  section_report("Summary")
  section_report("Fix")
  section_report("Verification")
  section_report("Related")

  body = ""
  for (i = 1; i <= line_count; i++) {
    scan_line(masked[i])
    body = body masked[i] "\n"
  }
  for (i = 1; i <= negated_count; i++) {
    print "negated\t" negated_order[i] "\t" negated_trigger[negated_order[i]]
  }
  if (has_closing) print "closing"
  if (has_non_closing) print "non-closing"
  if (tolower(body) ~ /(^|[^a-z0-9_])no (linked|related) issue([^a-z0-9_]|$)/) print "no-issue"
}
'
}

section_guidance() {
  case "$1" in
    Summary) echo 'Describe what this PR changes and why, in a sentence or two.' ;;
    Fix) echo 'State the concrete change and how it addresses the problem.' ;;
    Verification) echo 'Record concrete evidence the change works (commands, gates, output).' ;;
    Related) echo 'List related PRs, ADRs, or decision-log entries this PR does not close.' ;;
    *) echo 'Fill this section in.' ;;
  esac
}

linkage_result=pass
linkage_errors=()

ref_hint='#N'
pattern_note=''
keyword_kind=native
issue_kind='GitHub issue'
negation_reason="GitHub's linkage parser ignores the surrounding words, so this still registers a closing reference and still auto-closes the issue when this PR merges."
if [[ "$ISSUE_REF_PATTERN" != "$DEFAULT_ISSUE_REF_PATTERN" ]]; then
  ref_hint='<ref>'
  pattern_note=" Here <ref> must match this repository's issue-ref-pattern \`${ISSUE_REF_PATTERN}\`."
  keyword_kind=recognized
  issue_kind='tracked issue'
  negation_reason='Keyword-based closing parsers ignore the surrounding words, so this still reads as a closing reference.'
fi

is_exempt_author=false
if [[ -n "${EXEMPT_AUTHORS//[[:space:]]/}" && -n "$pr_author" ]]; then
  while IFS= read -r candidate; do
    candidate="${candidate#"${candidate%%[![:space:]]*}"}"
    candidate="${candidate%"${candidate##*[![:space:]]}"}"
    [[ -n "$candidate" ]] || continue
    # Exact-login equality only, never a `*[bot]` pattern, so an unknown future
    # bot is not silently skipped on this gate.
    if [[ "$candidate" == "$pr_author" ]]; then
      is_exempt_author=true
      break
    fi
    # `printf '%s\n'` (not '%s'): without the trailing newline `read` drops the
    # last list entry, which silently un-exempts the final author.
  done < <(printf '%s\n' "$EXEMPT_AUTHORS" | tr ',' '\n')
fi

if [[ "$is_exempt_author" == true ]]; then
  linkage_result=exempt
  echo "::notice::check-contract: PR author \"$(escape_annotation "$pr_author")\" matches an exempt-authors entry; skipping the issue-linkage check."
else
  analysis="$scratch/analysis.txt"
  analyze_body <"$scratch/body.txt" >"$analysis"

  negated_quoted=""
  while IFS=$'\t' read -r kind text trigger; do
    case "$kind" in
      section-missing) linkage_errors+=("Missing a \"## ${text}\" section. $(section_guidance "$text")") ;;
      section-empty) linkage_errors+=("The \"## ${text}\" section is empty.") ;;
      negated) negated_quoted+="${negated_quoted:+, }\"${text}\" (trigger \"${trigger}\")" ;;
      *) ;;
    esac
  done <"$analysis"

  if [[ -n "$negated_quoted" ]]; then
    linkage_errors+=("Negated closing reference (${negated_quoted}). ${negation_reason} Remove the closing keyword and use \"Refs: ${ref_hint}\" (or \"Relates to: ${ref_hint}\") on its own line instead.${pattern_note}")
  fi

  if ! grep -qxE 'closing|non-closing|no-issue' "$analysis"; then
    linkage_errors+=("Missing a ${keyword_kind} closing keyword (Closes/Fixes/Resolves ${ref_hint}). If this PR references an issue it must not close, put \"Refs: ${ref_hint}\" (or \"Relates to: ${ref_hint}\") on its own line. If it relates to no ${issue_kind} at all, state \"No linked issue\" (or \"No related issue:\") in the body instead.${pattern_note}")
  fi

  if [[ ${#linkage_errors[@]} -gt 0 ]]; then
    linkage_result=fail
  fi
fi

# ---------------------------------------------------------------------------
# Step 5 — advisory outcome: one marker comment, upserted, plus the label.
# ---------------------------------------------------------------------------
# Each call below is sent once: the outcome is advisory, and resending its
# calls could push the step past a short ci-status job's timeout.
GH_API_RETRY_ERRORS=false
# Sets `marker_comment_id` (empty when there is none), from the comments step 1
# read when that page held them all, else from the REST listing. Never call it in a
# command substitution: gh_api prints its warnings to stdout and adds its
# rate-limit wait to RATE_LIMIT_WAITED, and a subshell would capture the one
# and drop the other.
marker_comment_id=""
find_marker_comment() {
  local listing="$comments_json"
  marker_comment_id=""
  if [[ ! -f "$listing" ]]; then
    # shellcheck disable=SC2310 # gh_api handles its own errexit; the caller classifies the status.
    if ! gh_api GET "repos/${REPOSITORY}/issues/${PR_NUMBER}/comments" --paginate --slurp; then
      return 1
    fi
    listing="$gh_stdout"
  fi
  # Bot-authored only, newest first. On a public repository anyone can comment,
  # so a stranger who plants the marker would otherwise capture the upsert and
  # the gate would edit their comment instead of posting its own. `--slurp`
  # wraps every page in one outer array, so `max_by` sees all pages at once
  # rather than printing one id per page.
  marker_comment_id="$(jq -r --arg marker "$LINKAGE_MARKER" \
    '[ .[][] | select((.user.type // "") == "Bot" and ((.body // "") | contains($marker))) ] | (max_by(.id).id // empty)' \
    <"$listing")"
}

# shellcheck disable=SC2329 # invoked indirectly through best_effort.
upsert_marker_comment() {
  local body="$1"
  # shellcheck disable=SC2310 # find_marker_comment reports the read failure through its status.
  if ! find_marker_comment; then
    return 1
  fi
  jq -n --arg body "$body" '{body: $body}' >"$scratch/comment-payload.json"
  if [[ -n "$marker_comment_id" ]]; then
    gh_api PATCH "repos/${REPOSITORY}/issues/comments/${marker_comment_id}" --input "$scratch/comment-payload.json"
  else
    gh_api POST "repos/${REPOSITORY}/issues/${PR_NUMBER}/comments" --input "$scratch/comment-payload.json"
  fi
}

# shellcheck disable=SC2329 # invoked indirectly through best_effort.
add_linkage_label() {
  jq -n --arg label "$LINKAGE_LABEL" '{labels: [$label]}' >"$scratch/label-payload.json"
  gh_api POST "repos/${REPOSITORY}/issues/${PR_NUMBER}/labels" --input "$scratch/label-payload.json"
}

# shellcheck disable=SC2329 # invoked indirectly through best_effort.
remove_linkage_label() {
  # Percent-encoded so the label stays one path segment.
  gh_api DELETE "repos/${REPOSITORY}/issues/${PR_NUMBER}/labels/$(url_encode "$LINKAGE_LABEL")"
}

comment_heading="${LINKAGE_MARKER}"$'\n\n''**PR body contract — issue linkage**'$'\n\n'
if [[ "$linkage_result" == fail ]]; then
  annotation_level=warning
  if [[ "$LINKAGE_MODE" == enforce ]]; then
    annotation_level=error
  fi
  comment_body="${comment_heading}This PR body does not yet satisfy the issue-linkage contract:"$'\n'
  for message in "${linkage_errors[@]}"; do
    # The linkage messages quote body text (a negated closing reference and
    # its trigger word), so they are escaped like the title.
    echo "::${annotation_level}::check-contract: $(escape_annotation "$message")"
    comment_body+=$'\n'"- ${message}"
  done
  comment_body+=$'\n\n''Edit the body and this comment updates itself on the next run.'

  best_effort 'upserting the linkage comment' upsert_marker_comment "$comment_body"
  best_effort 'adding the linkage label' add_linkage_label
else
  if [[ "$linkage_result" == pass ]]; then
    echo 'check-contract: the PR body satisfies the issue-linkage contract.'
  fi
  # Never delete the marker comment: editing it keeps the audit trail of what
  # the gate said and why it stopped saying it.
  # shellcheck disable=SC2310 # a failed read simply means there is no comment to update.
  if find_marker_comment 2>/dev/null && [[ -n "$marker_comment_id" ]]; then
    best_effort 'updating the linkage comment' upsert_marker_comment "${comment_heading}This PR body conforms to the issue-linkage contract. Nothing to do."
  fi
  if grep -qxF -- "$LINKAGE_LABEL" "$scratch/labels.txt"; then
    best_effort 'removing the linkage label' remove_linkage_label
  fi
fi

# ---------------------------------------------------------------------------
# Step 6 — outputs and the single exit decision.
#
# Every check runs before any exit so all four outputs are always emitted; an
# early exit on the title would leave `linkage` unset for the caller.
# ---------------------------------------------------------------------------
emit_results "$title_result" "$do_not_merge_result" "$linkage_result" "$hold_result"

if [[ "$linkage_result" == fail && "$LINKAGE_MODE" == advisory ]]; then
  echo '::notice::check-contract: issue linkage is advisory in this repository; the step does not fail on it.'
fi
if [[ "$title_result" == fail || "$do_not_merge_result" == fail || "$hold_result" == fail ]] ||
  [[ "$linkage_result" == fail && "$LINKAGE_MODE" == enforce ]]; then
  exit 1
fi
